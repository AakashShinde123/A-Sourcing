     9→// ── seeded RNG (mulberry32) ─────────────────────────────────────
    10→let s = 42
    11→function rnd() {
    12→  s |= 0; s = (s + 0x6D2B79F5) | 0
    13→  let t = Math.imul(s ^ (s >>> 15), 1 | s)
    14→  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    15→  return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    16→}
    17→const pick = <T>(arr: T[]): T => arr[Math.floor(rnd() * arr.length)]
    18→const ri = (min: number, max: number) => Math.floor(rnd() * (max - min + 1)) + min
    19→const chance = (p: number) => rnd() < p

    21→const D = (daysAgo: number, h = 10, m = 30) => {
    22→  const d = new Date('2026-09-06T00:00:00+05:30')
    23→  d.setDate(d.getDate() - daysAgo)
    24→  d.setHours(h, m, 0, 0)
    25→  return d
    26→}
